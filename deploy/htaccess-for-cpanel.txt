# Optional .htaccess for the cPanel host.
# Copy this file into public_html and rename it to  .htaccess
# Nothing here is required — the site works without it.

DirectoryIndex index.php

# --- Serve /ebook as well as /ebook.php -------------------------------------
<IfModule mod_rewrite.c>
    RewriteEngine On

    # Send /ebook.php to /ebook (301), so only one URL is canonical
    RewriteCond %{THE_REQUEST} \s/+(.+?)\.php[\s?] [NC]
    RewriteRule ^ /%1 [R=301,L]

    # Serve /ebook from ebook.php
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteCond %{REQUEST_FILENAME}\.php -f
    RewriteRule ^(.*)$ $1.php [L]
</IfModule>

# --- Force HTTPS ------------------------------------------------------------
<IfModule mod_rewrite.c>
    RewriteCond %{HTTPS} !=on
    RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]
</IfModule>

# --- The partials are includes, not pages -----------------------------------
<FilesMatch "^(header|footer|counters)\.php$">
    Require all denied
</FilesMatch>

# --- The counter store is written by PHP; it should not be served -----------
# (data/.htaccess already denies it; this is belt and braces if that is lost)
RedirectMatch 404 ^/data/
RedirectMatch 404 ^/lib/

# --- Compression ------------------------------------------------------------
<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html text/css application/javascript image/svg+xml
</IfModule>

# --- Caching ----------------------------------------------------------------
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType text/css               "access plus 7 days"
    ExpiresByType application/javascript "access plus 7 days"
    ExpiresByType image/png              "access plus 30 days"
    ExpiresByType image/svg+xml          "access plus 30 days"
    ExpiresByType application/pdf        "access plus 30 days"
    ExpiresByType text/html              "access plus 0 seconds"
</IfModule>
